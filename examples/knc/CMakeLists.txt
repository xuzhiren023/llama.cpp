option(LLAMA_KNC_HOST_REFERENCE "Build a host reference runner; not KNC hardware validation" OFF)
if (NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The KNC runner requires Linux")
endif()
option(LLAMA_KNC_TESTS "Build offline KNC fixture helpers" OFF)
if (LLAMA_KNC_TESTS)
    add_executable(knc-fixture-quantize fixture-quantize.cpp)
    target_link_libraries(knc-fixture-quantize PRIVATE llama)
    add_executable(test-knc-mtmd test-image.cpp)
    target_link_libraries(test-knc-mtmd PRIVATE mtmd)
    target_include_directories(test-knc-mtmd PRIVATE ../../tests)
    if (GGML_KNC)
        target_compile_features(knc-fixture-quantize PRIVATE cxx_std_14)
        target_compile_features(test-knc-mtmd PRIVATE cxx_std_14)
        set_target_properties(test-knc-mtmd PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
    else()
        target_compile_features(knc-fixture-quantize PRIVATE cxx_std_17)
        target_compile_features(test-knc-mtmd PRIVATE cxx_std_17)
        add_executable(test-knc-mtmd-upstream ../../tests/test-mtmd-impl.cpp)
        target_compile_features(test-knc-mtmd-upstream PRIVATE cxx_std_17)
        target_link_libraries(test-knc-mtmd-upstream PRIVATE mtmd)
    endif()
endif()
if (NOT GGML_KNC AND NOT LLAMA_KNC_HOST_REFERENCE)
    message(FATAL_ERROR "Set GGML_KNC or explicitly select LLAMA_KNC_HOST_REFERENCE")
endif()
if (GGML_KNC AND LLAMA_KNC_HOST_REFERENCE)
    message(FATAL_ERROR "A KNC binary cannot be labelled as a host reference")
endif()
if (NOT TARGET mtmd)
    message(FATAL_ERROR "The KNC runner requires LLAMA_BUILD_MTMD=ON")
endif()
add_executable(llama-knc-runner knc.cpp)
target_link_libraries(llama-knc-runner PRIVATE llama mtmd vendor::stb)
if (TARGET OpenMP::OpenMP_CXX)
    target_link_libraries(llama-knc-runner PRIVATE OpenMP::OpenMP_CXX)
endif()
if (GGML_KNC)
    target_compile_features(llama-knc-runner PRIVATE cxx_std_14)
    target_include_directories(llama-knc-runner PRIVATE ../../ggml/src/ggml-cpu/arch/knc)
    target_compile_definitions(llama-knc-runner PRIVATE PHI_KNC_KERNELS PHI_PROFILE="${GGML_KNC_PROFILE}")
    set_target_properties(llama-knc-runner PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
else()
    target_compile_features(llama-knc-runner PRIVATE cxx_std_17)
    target_compile_definitions(llama-knc-runner PRIVATE PHI_HOST_REFERENCE)
endif()
